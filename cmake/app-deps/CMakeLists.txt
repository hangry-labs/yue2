cmake_minimum_required(VERSION 3.19)
project(Yue2AppDependencies LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
if(NOT MSVC)
    add_compile_definitions(_FORTIFY_SOURCE=2)
endif()
include(../app-deps.cmake)
install(FILES "${YUE2_MINIMP3_SOURCE}/minimp3.h" DESTINATION include)
foreach(package httplib yyjson minimp3)
    string(TOUPPER "${package}" upper)
    install(FILES "${YUE2_${upper}_SOURCE}/LICENSE" DESTINATION licenses RENAME "${package}.txt")
endforeach()
install(FILES "${CMAKE_CURRENT_LIST_DIR}/../../docker/app-deps.json" DESTINATION share/yue2-app-deps)
